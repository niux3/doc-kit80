src/controller/DocsController.js JavaScript (ES6) 
import { AppController } from './AppController'

export class DocsController extends AppController {
  async afterLoad(ctx) {
    // Interception & Guard avant rendu
    if (ctx.route.meta?.requiresAuth && !this.container.get('auth').check()) {
      return this.redirect('/login')
    }
  }

  async show(ctx) {
    const markdown = this.container.get('markdown')
    const docContent = await fetch(`/data/${ctx.params.slug}.md`).then(r => r.text())

    return this.render('docs/view', {
      content: markdown.parse(docContent)
    })
  }
}
